function parseDMY(str) {
if(!str) return null;
let parts = str.split("/");
if(parts.length < 3) return null;
return new Date(parts[2], parts[1] - 1, parts[0]);
}

function formatDMY(dateObj) {
if(!dateObj || isNaN(dateObj.getTime())) return "";
return `${String(dateObj.getDate()).padStart(2,'0')}/${String(dateObj.getMonth()+1).padStart(2,'0')}/${dateObj.getFullYear()}`;
}

function nextDay(dateObj) {
let next = new Date(dateObj);
next.setDate(next.getDate() + 1);
return next;
}

function convertLeavePointsToDays(pointsVal, ruleType) {
let base = Math.floor(pointsVal);
let decimals = Math.round((pointsVal % 1) * 100);
let divisor = (ruleType === "EL") ? 11 : 30;
let totalDays = (base * divisor) + decimals;
let convertedBase = Math.floor(totalDays / divisor);
let convertedDecimals = totalDays % divisor;
return convertedBase + (convertedDecimals / 100);
}

function applyLimits(val, maxVal) {
let base = Math.floor(val);

return base + (dec / 100);
}

function calculateBonusLeave() {
let startStr = document.getElementById('bonus_start_date').value;
let endStr = document.getElementById('bonus_end_date').value;
if(!startStr || !endStr) { alert("कृपया सुरुवातीचा आणि शेवटचा दिनांक निवडा!"); return; }

let dStart = parseDMY(startStr); let dEnd = parseDMY(endStr);
if(!dStart || !dEnd) { alert("कृपया वैध तारखा टाका!"); return; }

let totalCalendarDays = Math.round((dEnd.getTime() - dStart.getTime()) / (1000 * 60 * 60 * 24)) + 1;
let totalR = 0; let totalE = 0;

history.forEach((item, index) => {
if (item.type === "bonus" || item.q === "सरेंडर रजा" || item.type === "annual_extra") return;

let itemO = parseDMY(item.o); let itemP = parseDMY(item.p);
if(itemO && itemP && itemO >= dStart && itemP <= dEnd) {
totalR += (parseFloat(item.r_diff) || 0);
totalE += (parseFloat(item.e) || 0);
}
}
});

let dutyDays = totalCalendarDays - (totalR + totalE);
if(dutyDays < 0) dutyDays = 0;

let splitDate = parseDMY("01/04/1990");
let earnedBonusCalculated = 0;

if (dEnd < splitDate) {
earnedBonusCalculated = Math.floor((dutyDays * 10) / 365);
} else if (dStart >= splitDate) {
earnedBonusCalculated = Math.floor((dutyDays * 15) / 365);
} else {
let preDays = Math.round((splitDate.getTime() - dStart.getTime()) / (1000 * 60 * 60 * 24));
let postDays = totalCalendarDays - preDays;
let preRatio = preDays / totalCalendarDays;
let postRatio = postDays / totalCalendarDays;
let preDuty = dutyDays * preRatio;
let postDuty = dutyDays * postRatio;
earnedBonusCalculated = Math.floor(((preDuty * 10) / 365) + ((postDuty * 15) / 365));
}

tempBonusCalculated = earnedBonusCalculated;
tempBonusPeriod = startStr + ' ते ' + endStr;
tempTotalDays = totalCalendarDays; tempSpentDays = totalR; tempDutyDays = dutyDays;

}

if (duIdx < payMatrix[dueLevel].length - 1) activeDueBasic = payMatrix[dueLevel][duIdx + 1];
}
}
}
calculatedStatements.push({
month: mLabel, actualDays: daysInMonth, presentDays: daysInMonth,
daRate: getHistoricalDARate(currentCursor), hraRate: getHistoricalHRARate(currentCursor),
drBasicNominal: activeDrawnBasic, duBasicNominal: activeDueBasic,
drBasic: activeDrawnBasic, drDa: 0, drHra: 0, drCps: 0, drChs: 0, drTotal: 0,
duBasic: activeDueBasic, duDa: 0, duHra: 0, duCps: 0, duChs: 0, duTotal: 0,
diffBasic: 0, diffDa: 0, diffHra: 0, diffCps: 0, diffChs: 0, diffTotal: 0
});
currentCursor.setMonth(currentCursor.getMonth() + 1);
}
recalculateVolumeFiveMathLogic();
}

function recalculateVolumeFiveMathLogic() {
calculatedStatements.forEach(item => {
let daysInMonth = item.actualDays;
let pDays = item.presentDays;
let txt = item.month.toUpperCase().replace("-", " ");
let isMay = txt.includes("MAY") || txt.includes("मे");
let isJun = txt.includes("JUN") || txt.includes("जून");
let isJul = txt.includes("JUL") || txt.includes("जुलै");
let isYear2020 = txt.includes("20") || txt.includes("२०");

let isChsZeroPeriod = (isYear2020 && (isMay || isJun || isJul));

// DUE
item.duBasic = Math.round((item.duBasicNominal / daysInMonth) * pDays);
item.duDa = Math.round(item.duBasic * item.daRate);
item.duHra = Math.round(item.duBasic * item.hraRate);
item.duCps = Math.round((item.duBasic + item.duDa) * 0.10);
item.duChs = isChsZeroPeriod ? 0 : Math.round((item.duBasic + item.duDa) * 0.01);
item.duTotal = item.duBasic + item.duDa + item.duHra + item.duCps + item.duChs;

// DRAWN
item.drBasic = Math.round((item.drBasicNominal / daysInMonth) * pDays);
item.drDa = Math.round(item.drBasic * item.daRate);
item.drHra = Math.round(item.drBasic * item.hraRate);
item.drCps = Math.round((item.drBasic + item.drDa) * 0.10);
item.drChs = isChsZeroPeriod ? 0 : Math.round((item.drBasic + item.drDa) * 0.01);
item.drTotal = item.drBasic + item.drDa + item.drHra + item.drCps + item.drChs;

// DIFFERENCE
item.diffBasic = item.duBasic - item.drBasic;
item.diffDa = item.duDa - item.drDa;
item.diffHra = item.duHra - item.drHra;
item.diffCps = item.duCps - item.drCps;
item.diffChs = item.duChs - item.drChs;
item.diffTotal = item.duTotal - item.drTotal;
});
renderStatementTable();
}
